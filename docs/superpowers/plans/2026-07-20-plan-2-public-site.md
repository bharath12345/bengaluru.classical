# Plan 2 — Public Site (Read Paths) — Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Build the server-rendered public site with Django templates + HTMX + Tailwind v4 (Flowbite/Preline plain-HTML components), consuming the models from Plan 1. Deliver the complete read surface: upcoming/archive event lists with instant filtering, event detail pages with `schema.org/Event` JSON-LD, calendar view + subscribable ICS feed, series/festival evergreen pages, segmented sitemaps, and SEO/OpenGraph/discovery metadata.

**Architecture:** Pure Django SSR with HTMX for interactive filtering (no page reloads) and Tailwind v4 built via standalone CLI. The `apps.web` app holds all public views, templates, and URL patterns. Event queries flow through a custom manager (`Event.objects.published()`, `.upcoming()`, `.past()`) defined in `apps.events` so later plans (pipeline/MCP) share the same publishing rules. All public surfaces show ONLY `status=PUBLISHED` events. Mobile-first responsive design using Flowbite/Preline plain-HTML components. Calendar ICS feed uses `icalendar` library. Sitemaps via `django.contrib.sitemaps`.

**Tech Stack:** Django 5.x, HTMX 2.x (CDN), Tailwind CSS v4 standalone CLI, Flowbite/Preline (plain HTML), `icalendar`, `django.contrib.sitemaps`.

## Global Constraints

All constraints from Plan 1 apply, plus:
- **URL names/paths:** must match CONVENTIONS.md verbatim (`web:event_list` → `/`, `web:event_detail` → `/events/<slug>/`, `web:calendar` → `/calendar/`, `web:archive` → `/archive/`, `web:series_detail` → `/festivals/<slug>/`, `web:ics_feed` → `/calendar.ics`).
- **Publishing rules:** public pages expose ONLY `Event.objects.filter(status=Event.Status.PUBLISHED)`. Upcoming = published + `start_at >= now()`. Past = published + `start_at < now()`.
- **Manager/queryset methods** (`.published()`, `.upcoming()`, `.past()`) are added in Task 1 to `apps.events.models.EventQuerySet` so later plans consume them.
- **SEO:** `schema.org/Event` JSON-LD in initial HTML of every detail page with fields: `name`, `startDate` (ISO-8601 with +05:30 offset), `endDate`, `location` (Place + full address), `performer` (from EventArtist), `offers` (when ticketed, with `priceCurrency`), `eventStatus`, `image` (poster). OpenGraph/Twitter cards use the poster.
- **Template paths:** `templates/base.html`, `templates/web/event_list.html`, `event_detail.html`, `calendar.html`, `archive.html`, `series_detail.html`; partials under `templates/web/partials/` (`event_card.html`, `event_list_results.html` for HTMX swaps).
- **Static build:** Tailwind v4 standalone CLI builds to `app/static/dist/main.css`. Keep Node/npm out (use the single-binary CLI).
- **Mobile-first:** all layouts must adapt from 320px width upward.
- **No placeholders:** every code step shows complete code (no "TBD"/"similar to above").

import pytest
from django.utils import timezone

from apps.events.factories import EventFactory
from apps.events.models import Event

def test_published_filters_only_published():
    EventFactory(status=Event.Status.PUBLISHED)
    EventFactory(status=Event.Status.REVIEW)
    EventFactory(status=Event.Status.REJECTED)
    assert Event.objects.published().count() == 1


def test_upcoming_filters_published_and_future():
    now = timezone.now()
    EventFactory(status=Event.Status.PUBLISHED, start_at=now + timedelta(days=1))
    EventFactory(status=Event.Status.PUBLISHED, start_at=now - timedelta(days=1))
    EventFactory(status=Event.Status.REVIEW, start_at=now + timedelta(days=2))
    assert Event.objects.upcoming().count() == 1


def test_past_filters_published_and_past():
    now = timezone.now()
    EventFactory(status=Event.Status.PUBLISHED, start_at=now - timedelta(days=1))
    EventFactory(status=Event.Status.PUBLISHED, start_at=now + timedelta(days=1))
    EventFactory(status=Event.Status.REVIEW, start_at=now - timedelta(days=2))
    assert Event.objects.past().count() == 1


def test_upcoming_ordered_asc():
    now = timezone.now()
    e2 = EventFactory(status=Event.Status.PUBLISHED, start_at=now + timedelta(days=2))
    e1 = EventFactory(status=Event.Status.PUBLISHED, start_at=now + timedelta(days=1))
    upcoming = list(Event.objects.upcoming())
    assert upcoming == [e1, e2]

    now = timezone.now()
    e1 = EventFactory(status=Event.Status.PUBLISHED, start_at=now - timedelta(days=1))
    e2 = EventFactory(status=Event.Status.PUBLISHED, start_at=now - timedelta(days=2))

from django.db import models
from django.utils import timezone

from apps.core.models import Artist, City, Series, TimeStampedModel, Venue


class EventQuerySet(models.QuerySet):
    def published(self):
        return self.filter(status=Event.Status.PUBLISHED)

    def upcoming(self):
        return self.published().filter(start_at__gte=timezone.now()).order_by("start_at")

    def past(self):
        return self.published().filter(start_at__lt=timezone.now()).order_by("-start_at")


class EventManager(models.Manager):
    def get_queryset(self):
        return EventQuerySet(self.model, using=self._db)

    def published(self):
        return self.get_queryset().published()

    def upcoming(self):
        return self.get_queryset().upcoming()

    def past(self):
        return self.get_queryset().past()

# apps/web/views.py
from django.shortcuts import render

from apps.events.models import Event


def event_list(request):
    events = Event.objects.upcoming()

    # Filters
    genre = request.GET.get("genre")
    if genre:
        events = events.filter(genre=genre)

    free = request.GET.get("free")
    if free == "true":
        events = events.filter(is_free=True)

    area = request.GET.get("area")
    if area:
        events = events.filter(venue__area__icontains=area)

    # HTMX partial swap
    template = (
        "web/partials/event_list_results.html" if request.headers.get("HX-Request") else "web/event_list.html"
    )

    return render(request, template, {"events": events, "filters": request.GET})
```

# apps/web/views.py — APPEND this view
import json

from django.shortcuts import get_object_or_404, render

from apps.events.models import Event


def event_detail(request, slug):
    event = get_object_or_404(Event.objects.published(), slug=slug)

    # Build schema.org/Event JSON-LD
    jsonld = {
        "@context": "https://schema.org",
        "@type": "Event",
        "name": event.title,
        "startDate": event.start_at.isoformat(),
        "eventStatus": "https://schema.org/EventScheduled"
        if event.status == Event.Status.PUBLISHED
        else "https://schema.org/EventCancelled",
        "location": {
            "@type": "Place",
            "name": event.venue.name,
            "address": {
                "@type": "PostalAddress",
                "addressLocality": event.city.name,
                "addressRegion": "Karnataka",
                "addressCountry": "IN",
                "streetAddress": event.venue.address or event.venue.area,
            },
        },
    }

    if event.end_at:
        jsonld["endDate"] = event.end_at.isoformat()

    if event.event_artists.exists():
        jsonld["performer"] = [
            {"@type": "Person", "name": ea.artist.name} for ea in event.event_artists.all()
        ]

    if event.is_free:
        jsonld["isAccessibleForFree"] = True
    elif event.price:
        jsonld["offers"] = {
            "@type": "Offer",
            "price": str(event.price),
            "priceCurrency": event.currency,
            "url": event.source_url or "",
        }

    if event.poster:
        jsonld["image"] = request.build_absolute_uri(event.poster.url)

    if event.description:
        jsonld["description"] = event.description

    return render(
        request, "web/event_detail.html", {"event": event, "jsonld": json.dumps(jsonld, indent=2)}
    )
```

# apps/web/feeds.py
from django.http import HttpResponse
from icalendar import Calendar, Event as ICalEvent

from apps.events.models import Event


def ics_feed(request):
    events = Event.objects.upcoming()

    genre = request.GET.get("genre")
    if genre:
        events = events.filter(genre=genre)

    cal = Calendar()
    cal.add("prodid", "-//Bengaluru Classical//EN")
    cal.add("version", "2.0")
    cal.add("x-wr-calname", "Bengaluru Classical Concerts")

    for event in events:
        ical_event = ICalEvent()
        ical_event.add("summary", event.title)
        ical_event.add("dtstart", event.start_at)
        if event.end_at:
            ical_event.add("dtend", event.end_at)
        ical_event.add(
            "location", f"{event.venue.name}, {event.venue.area}, {event.city.name}".strip(", ")
        )
        if event.description:
            ical_event.add("description", event.description)
        ical_event.add("uid", f"event-{event.id}@bengaluruclassical.in")
        ical_event.add("dtstamp", event.created_at)
        cal.add_component(ical_event)

    response = HttpResponse(cal.to_ical(), content_type="text/calendar; charset=utf-8")
    response["Content-Disposition"] = 'attachment; filename="bengaluru-classical.ics"'
    return response
```

# apps/web/tests/test_integration.py
from datetime import timedelta

import pytest
from django.test import Client
from django.urls import reverse
from django.utils import timezone

from apps.core.factories import SeriesFactory
from apps.events.factories import EventFactory
from apps.events.models import Event

pytestmark = pytest.mark.django_db


def test_end_to_end_flow(client: Client):
    # Setup: published upcoming event, past event, series
    now = timezone.now()
    series = SeriesFactory(slug="ramanavami")
    upcoming = EventFactory(
        status=Event.Status.PUBLISHED,
        start_at=now + timedelta(days=1),
        slug="upcoming-concert",
        series=series,
    )
    past = EventFactory(
        status=Event.Status.PUBLISHED, start_at=now - timedelta(days=1), slug="past-concert"
    )

    # Step 1: Visit home/upcoming list
    response = client.get(reverse("web:event_list"))
    assert response.status_code == 200
    assert upcoming.title in response.content.decode()

    # Step 2: Click into detail page
    response = client.get(reverse("web:event_detail", args=[upcoming.slug]))
    assert response.status_code == 200
    assert "application/ld+json" in response.content.decode()

    # Step 3: Visit archive
    response = client.get(reverse("web:archive"))
    assert response.status_code == 200
    assert past.title in response.content.decode()

    # Step 4: Visit series page
    response = client.get(reverse("web:series_detail", args=[series.slug]))
    assert response.status_code == 200
    assert series.name in response.content.decode()

    # Step 5: Download ICS feed
    response = client.get(reverse("web:ics_feed"))
    assert response.status_code == 200
    assert b"BEGIN:VCALENDAR" in response.content

    # Step 6: Sitemap includes upcoming event
    response = client.get("/sitemap-upcoming.xml")
    assert upcoming.slug in response.content.decode()
```
